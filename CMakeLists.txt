cmake_minimum_required(VERSION 3.16)
project(aim_2025 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(casadi CONFIG REQUIRED)
add_executable(aim_demo src/main.cpp src/scenario.cpp src/simulation.cpp src/corridors.cpp src/optimization.cpp)
target_include_directories(aim_demo PRIVATE include)
target_link_libraries(aim_demo PRIVATE casadi::casadi)
if(MINGW)
 target_link_options(aim_demo PRIVATE -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME crossing_cluster COMMAND aim_demo --output validation)
